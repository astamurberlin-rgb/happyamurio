'use strict';
const test=require('node:test'),assert=require('node:assert/strict');
const {fixture,cloud,blank,pop}=require('./recovery-fixture.cjs');
const LIMIT=512*1024,SPARE=256*1024;
const archive=d=>JSON.parse(d.mem.get(d.c.HappyProfile.key('hd_journal_safety_v1'))||'null');
const size=d=>(d.mem.get(d.c.HappyProfile.key('hd_journal_safety_v1'))||'').length;
const usage=d=>[...d.mem].reduce((n,[k,v])=>n+k.length+v.length,0);
const probeKey=d=>d.c.HappyProfile.key('hd_journal_safety_v1:space-probe-v2');
function decode(v,id){const n=v.nodes[id];return n[0]==='v'?n[1]:n[0]==='a'?n[1].map(r=>decode(v,r)):Object.fromEntries(n[1].map(([k,r])=>[k,decode(v,r)]));}
async function edit(d,note){d.c.DB.days['2026-11-04'].note=note;await d.c.rawSet('hd_db_v1',JSON.stringify(d.c.DB));await d.run();}

test('100 KB plus 10 edits stays bounded and uploaded (original reviewer workload)',async()=>{
 cloud.clear();const d=await fixture(pop('x'.repeat(100000)),'A',{capacityChars:2*1024*1024});
 for(let i=0;i<10;i++){await edit(d,'x'.repeat(100000)+i);assert.equal(d.c.DayJournal.status().state,'uploaded');}
 assert.equal(archive(d).entries.filter(e=>e.state==='acknowledged').length,2);assert.ok(size(d)<320000);assert.ok(size(d)<LIMIT);assert.equal(d.mem.has(probeKey(d)),false);
 console.log('REPRO fixed:',JSON.stringify({recoveryChars:size(d),liveChars:d.mem.get(d.c.HappyProfile.key('hd_db_v1')).length,status:d.c.DayJournal.status().state}));
});

test('250 successive 100 KB edits keep syncing with two acknowledged checkpoints',async()=>{
 cloud.clear();const d=await fixture(pop('a'.repeat(100000)),'A',{capacityChars:2*1024*1024});let max=0;
 for(let i=0;i<250;i++){await edit(d,'a'.repeat(100000)+i);assert.equal(d.c.DayJournal.status().state,'uploaded');max=Math.max(max,size(d));assert.ok(size(d)<LIMIT);assert.ok(archive(d).entries.length<=2);assert.equal(d.mem.has(probeKey(d)),false);}
 assert.ok(d.opts.capacityChars-usage(d)>SPARE);console.log('LONG_UPLOAD:',JSON.stringify({edits:250,maxRecoveryChars:max,finalRecoveryChars:size(d)}));
});

test('120 alternating device edits/restores rotate only acknowledged history',async()=>{
 cloud.clear();const a=await fixture(pop('b'.repeat(100000)),'A',{capacityChars:3*1024*1024}),b=await fixture(blank(),'A',{capacityChars:3*1024*1024});let max=0;
 for(let i=0;i<120;i++){const writer=i%2?a:b,reader=i%2?b:a;await edit(writer,'b'.repeat(100000)+i);assert.equal(writer.c.DayJournal.status().state,'uploaded');await reader.run();assert.equal(reader.c.DayJournal.status().state,'restored');assert.equal(reader.c.DB.days['2026-11-04'].note,writer.c.DB.days['2026-11-04'].note);for(const d of [a,b]){max=Math.max(max,size(d));assert.ok(size(d)<LIMIT);const v=archive(d);assert.ok(v.entries.filter(e=>e.state==='acknowledged'&&e.action==='restore').length<=1);assert.ok(v.entries.filter(e=>e.state==='acknowledged'&&e.action!=='restore').length<=2);}}
 console.log('ALTERNATING:',JSON.stringify({edits:120,restores:120,maxRecoveryChars:max,finalA:size(a),finalB:size(b)}));
});

test('unresolved conflict bytes and each preserved unique pair remain pinned at cap',async()=>{
 cloud.clear();const remote=await fixture(pop('remote'));
 const d=await fixture(pop('c'.repeat(80000)+'initial'),'A',{capacityChars:3*1024*1024});assert.equal(d.c.DayJournal.status().state,'conflict');
 const first=archive(d),firstLocal=decode(first,first.entries[0].local);let stopped=false,lastRaw;
 for(let i=0;i<20;i++){lastRaw=d.mem.get(d.c.HappyProfile.key('hd_journal_safety_v1'));await edit(d,String.fromCharCode(65+i).repeat(80000)+i);const state=d.c.DayJournal.status().state;if(state==='error'){assert.equal(d.c.DayJournal.status().reason,'recovery-capacity');assert.equal(d.mem.get(d.c.HappyProfile.key('hd_journal_safety_v1')),lastRaw);stopped=true;break;}assert.equal(state,'conflict');}
 assert.equal(stopped,true);const v=archive(d);assert.ok(v.entries.every(e=>e.state==='unresolved'));assert.deepEqual(decode(v,v.entries[0].local),firstLocal);assert.ok(size(d)<=LIMIT);assert.equal(remote.c.DB.days['2026-11-04'].note,'remote');
 // Capacity refusal must leave enough room for a normal next local edit.
 d.c.DB.days['2026-11-04'].note+=' locally saved after conflict cap';await d.c.rawSet('hd_db_v1',JSON.stringify(d.c.DB));assert.ok(d.mem.get(d.c.HappyProfile.key('hd_db_v1')).includes('locally saved after conflict cap'));assert.equal(d.mem.has(probeKey(d)),false);
});

test('capacity preflight fails before adding backup or writing cloud, preserving ordinary-save room',async()=>{
 cloud.clear();const d=await fixture(pop('q'.repeat(100000)),'A',{capacityChars:500000});assert.equal(d.c.DayJournal.status().reason,'recovery-capacity');assert.equal(cloud.size,0);assert.equal(archive(d),null);assert.equal(d.mem.has(probeKey(d)),false);d.c.DB.days['2026-11-04'].note+='local';await d.c.rawSet('hd_db_v1',JSON.stringify(d.c.DB));assert.ok(usage(d)<110000);
});

test('headroom reserve accounts for next DB growth, not only the backup bytes',async()=>{
 cloud.clear();const d=await fixture(pop('headroom'),'A',{capacityChars:2*1024*1024});const before=d.mem.get(d.c.HappyProfile.key('hd_journal_safety_v1'));d.opts.capacityChars=usage(d)+10000;d.c.DB.days['2026-11-04'].note+='new';await d.c.rawSet('hd_db_v1',JSON.stringify(d.c.DB));const remoteBefore=JSON.stringify(cloud.get('A'));await d.run();assert.equal(d.c.DayJournal.status().reason,'recovery-capacity');assert.equal(d.mem.get(d.c.HappyProfile.key('hd_journal_safety_v1')),before);assert.equal(JSON.stringify(cloud.get('A')),remoteBefore);assert.equal(d.mem.has(probeKey(d)),false);
});

test('stale temporary probe is removed on next ordinary save and startup, leaving all recovery checkpoints',async()=>{
 cloud.clear();const d=await fixture(pop('safe'));const before=d.mem.get(d.c.HappyProfile.key('hd_journal_safety_v1'));d.mem.set(probeKey(d),'day-recovery-space-v2\n'+'0'.repeat(300000));d.c.DB.days['2026-11-04'].note+='edit';await d.c.rawSet('hd_db_v1',JSON.stringify(d.c.DB));assert.equal(d.mem.has(probeKey(d)),false);assert.equal(d.mem.get(d.c.HappyProfile.key('hd_journal_safety_v1')),before);
 d.mem.set(probeKey(d),'day-recovery-space-v2\n'+'0'.repeat(300000));const reloaded=await fixture(JSON.parse(d.mem.get(d.c.HappyProfile.key('hd_db_v1'))),'A',{mem:d.mem,offline:true});assert.equal(reloaded.mem.has(probeKey(d)),false);assert.equal(reloaded.mem.get(d.c.HappyProfile.key('hd_journal_safety_v1')),before);
});

test('post-commit compaction failure keeps current data and pending recovery; later retry acknowledges safely',async()=>{
 cloud.clear();const d=await fixture(pop('committed safely'),'A',{failAck:1});assert.equal(d.c.DayJournal.status().state,'uploaded');assert.equal(d.c.DB.days['2026-11-04'].note,'committed safely');assert.equal(archive(d).entries.at(-1).state,'pending');assert.match(d.c.DayJournal.status().message,/неподтверждённая/);
 await d.run();assert.equal(d.c.DayJournal.status().state,'current');assert.ok(archive(d).entries.every(e=>e.state==='acknowledged'));assert.equal(d.c.DB.days['2026-11-04'].note,'committed safely');
});

test('uncertain cloud outcome is pinned until a fresh read confirms that exact upload',async()=>{
 cloud.clear();let fail=true;const d=await fixture(pop('unknown result'),'A',{afterWrite:()=>{if(fail)throw Error('response lost')}});assert.equal(d.c.DayJournal.status().cloudWriteUncertain,true);assert.equal(archive(d).entries[0].state,'pending');fail=false;await d.run();assert.equal(d.c.DayJournal.status().state,'current');assert.ok(archive(d).entries.every(e=>e.state==='acknowledged'));
});

test('legacy archive conversion preserves all unresolved snapshots but drops duplicated revision tokens',async()=>{
 cloud.clear();const mem=new Map(),oldDB=pop('legacy'),evidence={action:'conflict',reason:'different-without-baseline',local:{snapshot:{app:'happyday',v:1,db:oldDB},revision:JSON.stringify([oldDB,'x'.repeat(1000000)])},remote:{snapshot:{app:'happyday',v:1,db:pop('legacy remote')},version:'legacy'},baseline:null};mem.set('owner:A:hd_journal_safety_v1',JSON.stringify([{at:1,rawDB:JSON.stringify(oldDB),inMemoryDB:oldDB,evidence}]));const d=await fixture(oldDB,'A',{mem,offline:true});d.opts.offline=false;await d.run();const v=archive(d);assert.equal(v.v,2);assert.ok(v.entries.some(e=>e.legacyUnconfirmed&&e.state==='unresolved'));assert.ok(v.entries.some(e=>decode(v,e.local).db.days['2026-11-04'].note==='legacy'));assert.ok(size(d)<10000);assert.equal(JSON.stringify(v).includes('revision'),false);
});
